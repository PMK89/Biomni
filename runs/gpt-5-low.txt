def find_human_pathway_id(term):
    res = query_reactome(prompt=f"Find the human Reactome pathway ID for: {term}")
    print(f"Search results for '{term}':")
    print(res)
    # Heuristic: fetch first human 'Pathway' entry if provided in result
    pathway_id = None
    if res.get('success') and 'result' in res and 'results' in res['result'] and res['result']['results']:
        entries = res['result']['results'][0].get('entries', [])
        for e in entries:
            if e.get('exactType') == 'Pathway' and 'Homo sapiens' in e.get('species', []):
                pathway_id = e.get('stId') or e.get('id')
                break
    return pathway_id

def get_pathway_reference_genes(pathway_id):
    if not pathway_id:
        return []
    endpoint = f"https://reactome.org/ContentService/data/participants/{pathway_id}/referenceEntities"
    res = query_reactome(prompt=None, endpoint=endpoint)
    print(f"Reference entities for {pathway_id}:")
    print(res)
    genes = set()
    if res.get('success') and 'result' in res and res['result']:
        # Reactome returns a list of ReferenceEntities with gene names in 'displayName' or 'geneName' fields
        for ent in res['result']:
            # Try multiple fields
            g = ent.get('geneName') or ent.get('displayName') or ent.get('name')
            if isinstance(g, list):
                for gg in g:
                    if isinstance(gg, str):
                        genes.add(gg.split(' ')[0])
            elif isinstance(g, str):
                genes.add(g.split(' ')[0])
            # Also extract from 'databaseName' or crossRefs if present
            if 'geneNames' in ent and isinstance(ent['geneNames'], list):
                for gg in ent['geneNames']:
                    if isinstance(gg, str):
                        genes.add(gg.split(' ')[0])
    return sorted(genes)
