# Helpers to query Reactome API via the provided tool
def reactome_search(term):
    endpoint = f"https://reactome.org/ContentService/search/query?query={term}&species=Homo%20sapiens"
    res = query_reactome(prompt=None, endpoint=endpoint, verbose=True)
    print(f"\n[SEARCH] Term: {term}")
    print(f"Status: {res.get('success')}, Endpoint: {endpoint}")
    return res

def parse_search_results(res, name_filters=None, type_filter="Pathway", max_hits=3):
    # Reactome tool returns dict with 'result' -> 'results'
    results = res.get("result", {}).get("results", [])
    out = []
    for item in results:
        # item can have nested 'entries' for each matched pathway
        entries = item.get("entries", [])
        for e in entries:
            ename = e.get("name", "")
            etype = e.get("exactType", e.get("type", ""))
            species = e.get("species", [])
            if type_filter in etype and ("Homo sapiens" in species):
                if name_filters is None or any(f.lower() in ename.lower() for f in name_filters):
                    out.append({
                        "name": ename,
                        "stId": e.get("stId") or e.get("id"),
                        "dbId": e.get("dbId"),
                        "type": etype,
                        "species": species
                    })
    # Deduplicate by stId and keep first N
    seen = set()
    uniq = []
    for o in out:
        sid = o["stId"]
        if sid and sid not in seen:
            seen.add(sid)
            uniq.append(o)
    uniq = uniq[:max_hits]
    print(f"[PARSE] Filtered pathway hits: {len(uniq)}")
    for u in uniq:
        print(f"  - {u['stId']} | {u['name']}")
    return uniq

    for it in items if isinstance(items, list) else []:
        # Participants can be EntityWithAccessionedSequence with referenceEntity details
        ref = it.get("referenceEntity") or {}
        dbname = ref.get("databaseName") or ref.get("database") or ""
        species = it.get("speciesName") or it.get("species", "")
        cls = it.get("className", "")
        # filter for human protein entities with UniProt mapping
        if "Homo sapiens" in str(species) and "EntityWithAccessionedSequence" in str(cls) and "UniProt" in str(dbname):
            # try gene name fields
            g = ref.get("geneName")
            if isinstance(g, list):
                for gg in g:
                    symbols.add(gg)

                symbols.add(g)
            else:
                # fallback to displayName if looks like a gene
                dn = it.get("displayName", "")
                if dn and dn.isupper() and len(dn) <= 10:
                    symbols.add(dn)
    print(f"[EXTRACT] Extracted {len(symbols)} putative gene symbols")
    return sorted(symbols)
