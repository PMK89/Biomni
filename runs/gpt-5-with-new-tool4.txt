def parse_gene_symbols_literal(val):
    # Convert MSigDB 'geneSymbols' column to a clean list of symbols
    if val is None:
        return []
    if isinstance(val, (list, tuple, set)):
        items = list(val)
    elif isinstance(val, str):
        # If the string looks like a Python list, try literal_eval
        s = val.strip()
        if len(s) > 0 and s[0] == '[' and s[-1] == ']':
            try:
                le = ast.literal_eval(s)
                items = le if isinstance(le, (list, tuple, set)) else [s]
            except Exception:
                # Fallback split
                items = re.split(r"[,;\s]+", s)
        else:
            items = re.split(r"[,;\s]+", s)
    else:
        items = []
    # Clean tokens, keep alphanumerics and dashes only
    cleaned = []
    for x in items:
        if x is None:
            continue
        xs = str(x)
        token = "".join(re.findall(r"[A-Za-z0-9\-]+", xs))
        token = token.upper().strip()
        if token and token != 'NAN':
            cleaned.append(token)
    return cleaned
